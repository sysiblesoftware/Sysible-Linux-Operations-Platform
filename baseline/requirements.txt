# Sysible Baseline — the SLOP config-management module runtime. Deliberately
# tiny: sqlite3, json, fnmatch and shlex are all stdlib, so the service stays
# easy to audit.
#
# NOTE: salt itself is NOT a dependency of this service. Baseline is masterless
# (see backend/salt.py) — it composes a `salt-call --local` command and the HOST
# runs it. Installing salt here would add a large dependency tree, a C toolchain
# for timelib, and no capability: this container never evaluates a state.
fastapi==0.115.6
uvicorn[standard]==0.34.0
# httpx is the one addition, and only for the console's "Apply now" button,
# which asks the CONTROLLER to run a state (Baseline never talks to hosts).
# backend/controller.py imports it DEFENSIVELY: if it is ever missing, the
# button turns itself off and says so, rather than taking the whole service
# down — which is exactly what happened to Flashback once when the import was
# added here before the dependency was.
httpx==0.28.1
